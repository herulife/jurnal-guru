function loadRekapNilai(){
  var kid=document.getElementById('rekapNilaiKelas').value,mp=document.getElementById('rekapNilaiMapel').value.trim().toLowerCase(),kat=document.getElementById('rekapNilaiKategori').value;
  sL();
  gs('apiGetNilai',A.token,{kelasId:kid||null}).then(function(r){
    hL();if(!r||!r.ok){toast('Gagal load nilai','er');return}
    var filtered=r.data.filter(function(n){if(mp&&(n.Mata_Pelajaran||'').toLowerCase().indexOf(mp)<0)return false;if(kat&&(n.Kategori||'')!==kat)return false;return true});
    RN.nil=filtered;
    buildRekapNilai();
  }).catch(function(e){hL();toast('Gagal: '+e.message,'er')})
}
function buildRekapNilai(){
  var sis={},bab={},cell={};
  for(var i=0;i<RN.nil.length;i++){
    var n=RN.nil[i],sid=n.Siswa_ID||'x'+i,bb=n.Bab||'Tanpa Bab';
    sis[sid]=n.Nama_Siswa||'-';
    bab[bb]=true;
    var key=sid+'|'+bb;
    if(!cell[key])cell[key]=[];
    cell[key].push(Number(n.Nilai)||0);
  }
  var siswaNames=Object.keys(sis).sort(function(a,b){return(sis[a]||'').localeCompare(sis[b]||'')});
  var babList=Object.keys(bab);
  RN.siswa=siswaNames;RN.babes=babList;RN.cell=cell;RN.sis=sis;
  renderRekapNilai();
}
function renderRekapNilai(){
  var thead=document.getElementById('theadRekapNilai'),tb=document.getElementById('tbodyRekapNilai'),info=document.getElementById('rekapNilaiInfo');
  if(!RN.siswa||!RN.siswa.length){
    tb.innerHTML='<tr><td colspan="3" class="text-center text-gray-400 py-8">Tidak ada data</td></tr>';
    thead.innerHTML='<tr><th>No</th><th>Nama Siswa</th><th>Bab</th></tr>';
    info.textContent='';return
  }
  var h='<tr><th rowspan="2">No</th><th rowspan="2">Nama Siswa</th>';
  for(var i=0;i<RN.babes.length;i++)h+='<th>'+esc(RN.babes[i])+'</th>';
  h+='<th>Rata-rata</th></tr><tr>';
  for(var i=0;i<RN.babes.length;i++)h+='<th class="font-normal text-[10px]">Nilai</th>';
  h+='<th class="font-normal text-[10px]">Total</th></tr>';
  thead.innerHTML=h;
  var body='';
  for(var s=0;s<RN.siswa.length;s++){
    var sid=RN.siswa[s],sum=0,cnt=0;
    body+='<tr><td>'+(s+1)+'</td><td class="font-semibold">'+esc(RN.sis[sid]||'-')+'</td>';
    for(var b=0;b<RN.babes.length;b++){
      var vals=RN.cell[sid+'|'+RN.babes[b]],v=vals&&vals.length?(vals.reduce(function(a,c){return a+c})/vals.length):'-';
      body+='<td>'+v+'</td>';

    }
    var avg=cnt>0?Math.round(sum/cnt*10)/10:'-';
    body+='<td class="font-bold">'+avg+'</td></tr>';
  }
  tb.innerHTML=body;
  info.textContent=RN.siswa.length+' siswa, '+RN.babes.length+' bab';
}

function parseCSVLine(line){
  var result=[],current='',inQuotes=false;
  for(var i=0;i<line.length;i++){
    var c=line[i];
    if(inQuotes){if(c==='"'&&line[i+1]==='"'){current+='"';i++}else if(c==='"'){inQuotes=false}else{current+=c}}
    else{if(c==='"'){inQuotes=true}else if(c===','){result.push(current.trim());current=''}else{current+=c}}
  }
  result.push(current.trim());return result;
}

function swapMember(el){
  var newSid=el.value,oldSid=el.getAttribute('data-sid'),g=el.getAttribute('data-grp');
  if(newSid===oldSid)return;
  // Remove old student from this slot
  for(var i=0;i<KB.kelompok.length;i++){if(KB.kelompok[i].Siswa_ID===oldSid&&KB.kelompok[i].Kelompok===g){KB.kelompok.splice(i,1);break}}
  // Remove the new student from wherever they currently are
  for(var i=KB.kelompok.length-1;i>=0;i--){if(KB.kelompok[i].Siswa_ID===newSid)KB.kelompok.splice(i,1)}
  // Find student data
  var ns=null;for(var i=0;i<KB.siswa.length;i++){if(KB.siswa[i].ID===newSid){ns=KB.siswa[i];break}}
  if(ns)KB.kelompok.push({Kelas_ID:document.getElementById('kbKelas').value,Kelompok:g,No:'',Siswa_ID:ns.ID,NIS:ns.NIS||'',Nama_Siswa:ns.Nama_Siswa,JK:ns.Jenis_Kelamin,Kelas_Asal:ns.Nama_Kelas||'',Nilai:ns.Nilai||''})
  renderKelompok()
}

function importExcelKelompok(input){
  if(!input.files||!input.files[0])return;
  var file=input.files[0];if(!file.name.match(/\.xlsx?$/i)){toast('Pilih file .xlsx','wr');return}
  document.getElementById('kbInfo').textContent='Membaca file...';
  var reader=new FileReader();
  reader.onload=function(e){
    try{
      var wb=XLSX.read(e.target.result,{type:'array'}),sheetNames=wb.SheetNames;
      var kelasMap={},sel=document.getElementById('kbKelas');
      for(var i=0;i<sel.options.length;i++){if(sel.options[i].value)kelasMap[sel.options[i].text.trim()]=sel.options[i].value}
      var allPromises=[],importCount=0;
      for(var si=0;si<sheetNames.length;si++){
        var sn=sheetNames[si],ws=wb.Sheets[sn],data=XLSX.utils.sheet_to_json(ws,{defval:''});
        if(!data.length)continue;
        var kelasId=kelasMap[sn]||kelasMap['X '+sn]||kelasMap[sn.replace(/^X\s*/,'')]||'';
        if(!kelasId){document.getElementById('kbInfo').textContent='Skip sheet "'+sn+'": tidak cocok dengan kelas';continue}
        var records=[];
        for(var i=0;i<data.length;i++){
          var r=data[i];
          records.push({
            Kelompok:String(r.Kelompok||r.kelompok||'').trim(),
            No:String(r.No||r.no||'').trim(),Siswa_ID:'',
            NIS:String(r.NIS||r.nis||'').trim(),
            Nama_Siswa:String(r.Nama_Siswa||r.nama_siswa||r.Nama||r.nama||'').trim(),
            JK:String(r.JK||r.jk||r.Jenis_Kelamin||r.jenis_kelamin||'').trim(),
            Kelas_Asal:String(r.Kelas_Asal||r.kelas_asal||sn).trim(),
            Nilai:String(r.Nilai||r.nilai||'').trim()
          })
        }
        if(records.length){importCount+=records.length;allPromises.push(gs('apiImportKelompokExcel',A.token,{kelasId:kelasId,records:records}))}
      }
      if(!allPromises.length){document.getElementById('kbInfo').textContent='Tidak ada data sheet yang cocok dengan kelas';input.value='';return}
      document.getElementById('kbInfo').textContent='Mengimport '+importCount+' data ke '+allPromises.length+' kelas...';
      Promise.all(allPromises).then(function(results){
        var ok=0;results.forEach(function(r){if(r&&r.ok)ok++});
        document.getElementById('kbInfo').textContent=importCount+' data diimport ('+ok+'/'+allPromises.length+' kelas berhasil)';
        toast(importCount+' data kelompok berhasil diimport');
        // Refresh current class
        loadKelompok()
      }).catch(function(err){
        document.getElementById('kbInfo').textContent='Gagal import';
        toast('Gagal: '+err.message,'er')
      })
    }catch(ex){document.getElementById('kbInfo').textContent='Gagal baca Excel';toast('Gagal baca Excel: '+ex.message,'er')}
    input.value=''
  };
  reader.readAsArrayBuffer(file)
}

/* === INIT === */
function init(){
  if(!A.token){document.getElementById('loginPage').classList.remove('hidden');document.getElementById('appPage').classList.add('hidden');return}
  gs('apiCheckAuth',A.token).then(function(r){
    if(r&&r.ok){A.user=r.user;enterApp()}
    else{A.token='';localStorage.removeItem('td_token');document.getElementById('loginPage').classList.remove('hidden');document.getElementById('appPage').classList.add('hidden')}
  }).catch(function(){A.token='';localStorage.removeItem('td_token');document.getElementById('loginPage').classList.remove('hidden');document.getElementById('appPage').classList.add('hidden')});
}
if(document.readyState==='loading')document.addEventListener('DOMContentLoaded',init);else init();
</script>
