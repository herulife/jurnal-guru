function apiUploadSiswaBatch(t, arr) {
  var s = getSession_(t);
  if (!s) return { ok: false, msg: 'Unauthorized' };
  try {
    var sh = getSheet_('Data_Siswa');
    var kl = getAllRows_('Data_Kelas');
    var kelasMap = {};
    for (var i = 0; i < kl.length; i++) kelasMap[kl[i].Nama_Kelas] = kl[i].ID;
    var added = 0, skipped = 0, errors = [];
    for (var i = 0; i < arr.length; i++) {
      var d = arr[i];
      if (!d.Nama_Siswa || !d.NIS) { skipped++; errors.push('Baris ' + (i + 2) + ': NIS dan Nama wajib diisi'); continue; }
      var kelasId = kelasMap[d.Nama_Kelas] || '';
      if (!kelasId && d.Nama_Kelas) { skipped++; errors.push('Baris ' + (i + 2) + ': Kelas "' + d.Nama_Kelas + '" tidak ditemukan'); continue; }
      sh.appendRow([uuid_(), d.NIS, d.NISN || '', d.Nama_Siswa, d.Jenis_Kelamin || 'L', kelasId, d.Alamat || '', d.Telepon || '', d.Email || '', d.Nama_Ortu || '']);
      added++;
    }
    addLog_(s.userId, 'UPLOAD_SISWA', 'Upload massal: ' + added + ' ditambah, ' + skipped + ' dilewati');
    return { ok: true, msg: added + ' siswa berhasil ditambahkan, ' + skipped + ' dilewati', added: added, skipped: skipped, errors: errors };
  } catch (e) { return { ok: false, msg: e.message }; }
}

function generatePdf_(title, headers, rows, meta) {
  var doc = DocumentApp.create('PDF_' + Date.now());
  var body = doc.getBody();
  body.setMarginLeft(50).setMarginRight(50).setMarginTop(40).setMarginBottom(40);
  if (meta.schoolName) {
    var p1 = body.appendParagraph(meta.schoolName);
    p1.setHeading(DocumentApp.ParagraphHeading.HEADING1).setAlignment(DocumentApp.TextAlignment.CENTER);
  }
  if (meta.address) body.appendParagraph(meta.address).setAlignment(DocumentApp.TextAlignment.CENTER);
  if (meta.npsn) body.appendParagraph('NPSN: ' + meta.npsn).setAlignment(DocumentApp.TextAlignment.CENTER);
  body.appendHorizontalRule();
  var tp = body.appendParagraph(title);
  tp.setHeading(DocumentApp.ParagraphHeading.HEADING2).setAlignment(DocumentApp.TextAlignment.CENTER);
  if (meta.filters) {
    for (var i = 0; i < meta.filters.length; i++) {
      body.appendParagraph(meta.filters[i]).setAlignment(DocumentApp.TextAlignment.LEFT);
    }
  }
  if (rows.length > 0) {
    var maxCols = headers.length;
    var tblData = [headers];
    for (var ri = 0; ri < rows.length; ri++) {
      var row = [];
      for (var ci = 0; ci < maxCols; ci++) {
        var val = (rows[ri][ci] !== null && rows[ri][ci] !== undefined) ? String(rows[ri][ci]) : '';
        row.push(val);
      }
      tblData.push(row);
    }
    var table = body.appendTable(tblData);
    for (var c = 0; c < maxCols; c++) {
      var cell = table.getRow(0).getCell(c);
      cell.setBold(true).setBackgroundColor('#1A2332');
      cell.editAsText().setForegroundColor('#ffffff').setFontSize(9);
    }
    for (var r = 1; r <= rows.length; r++) {
      for (var c = 0; c < maxCols; c++) {
        var tc = table.getRow(r).getCell(c);
        tc.setFontSize(9);
        if (r % 2 === 0) tc.setBackgroundColor('#f5f5f5');
      }
    }
  } else {
    body.appendParagraph('Tidak ada data untuk ditampilkan.');
  }
  body.appendParagraph('');
  body.appendParagraph('Dicetak pada: ' + new Date().toLocaleString('id-ID')).setFontSize(8);
  if (meta.printedBy) body.appendParagraph('Oleh: ' + meta.printedBy).setFontSize(8);
  doc.saveAndClose();
  var pdfBlob = DriveApp.getFileById(doc.getId()).getBlob().getAs('application/pdf');
  DriveApp.getFileById(doc.getId()).setTrashed(true);
  return Utilities.base64Encode(pdfBlob.getBytes());
}

function getProfil_() {

function fmtDateStr_(v) {
  if (!v) return '-';
  var d = new Date(v);
  return isNaN(d.getTime()) ? String(v) : d.toLocaleDateString('id-ID', { day: '2-digit', month: '2-digit', year: 'numeric' });
}

      }
      if (tot > 0 && (abs / tot) > 0.3) notif.push({ type: 'danger', msg: sw[i].Nama_Siswa + ' absensi ' + Math.round(abs/tot*100) + '% tidak hadir', icon: 'fa-user-graduate' });

// === KALENDER ===
function apiGetKalenderEvents(t, month, year) {
  var s = getSession_(t);
  if (!s) return { ok: false, msg: 'Unauthorized' };
  try {
    var events = [], jd = getAllRows_('Jadwal_Mengajar'), ab = getAllRows_('Absensi'), kl = getAllRows_('Data_Kelas');
    var hariMap = { 'Monday': 'Senin', 'Tuesday': 'Selasa', 'Wednesday': 'Rabu', 'Thursday': 'Kamis', 'Friday': 'Jumat', 'Saturday': 'Sabtu', 'Sunday': 'Minggu' };
    if (month && year) {
      var daysInMonth = new Date(year, month, 0).getDate();
      for (var d = 1; d <= daysInMonth; d++) {
        var dt = new Date(year, month - 1, d);
        var dayName = hariMap[dt.toLocaleDateString('en-US', { weekday: 'long' })];
        if (dayName === 'Minggu') continue;
        for (var i = 0; i < jd.length; i++) {
          if (jd[i].Hari === dayName) {
            var kN = '-';
            for (var j = 0; j < kl.length; j++) { if (kl[j].ID === jd[i].Kelas_ID) { kN = kl[j].Nama_Kelas; break; } }
            if (!kN) { var idx = Number(jd[i].Kelas_ID) - 1; if (idx >= 0 && idx < kl.length) kN = kl[idx].Nama_Kelas; }
            events.push({ date: year + '-' + String(month).padStart(2, '0') + '-' + String(d).padStart(2, '0'), title: jd[i].Mata_Pelajaran + ' - ' + kN, time: (jd[i].Jam_Mulai || '') + '-' + (jd[i].Jam_Selesai || ''), type: 'jadwal' });
          }
        }
      }
    }
    for (var i = 0; i < ab.length; i++) {
      var stN = '-';
      for (var j = 0; j < kl.length; j++) { if (kl[j].ID === ab[i].Kelas_ID) { stN = kl[j].Nama_Kelas; break; } }
      events.push({ date: ab[i].Tanggal, title: 'Absensi ' + stN + ' (' + ab[i].Status + ')', time: '', type: 'absensi' });
    }
    events.sort(function(a, b) { return String(a.date).localeCompare(String(b.date)); });
    return { ok: true, data: events.slice(0, 100) };
  } catch (e) { return { ok: false, msg: e.message }; }
}

function apiGetKelompok(t, fl) {

  if (!s) return { ok: false, msg: 'Unauthorized' };

    var kl = getAllRows_('Kelompok_Belajar'), ks = getAllRows_('Data_Kelas'), sw = getAllRows_('Data_Siswa');

  if (!s) return { ok: false, msg: 'Unauthorized' };
  if (!data || !data.kelasId || !data.records) return { ok: false, msg: 'Data tidak lengkap' };
  try {
    var sh = getSheet_('Kelompok_Belajar');
    if (!sh) { sh = SpreadsheetApp.openById(SPREADSHEET_ID).insertSheet('Kelompok_Belajar'); sh.appendRow(['ID','Kelas_ID','Kelompok','No','Siswa_ID','NIS','Nama_Siswa','JK','Kelas_Asal','Nilai']); }
    var ex = getAllRows_('Kelompok_Belajar');
    var toKeep = [];
    for (var i = 0; i < ex.length; i++) { if (ex[i].Kelas_ID !== data.kelasId) toKeep.push(ex[i]); }
    sh.clear();
    var hd = ['ID','Kelas_ID','Kelompok','No','Siswa_ID','NIS','Nama_Siswa','JK','Kelas_Asal','Nilai'];
    sh.appendRow(hd);
    for (var i = 0; i < toKeep.length; i++) { sh.appendRow([toKeep[i].ID,toKeep[i].Kelas_ID,toKeep[i].Kelompok,toKeep[i].No,toKeep[i].Siswa_ID,toKeep[i].NIS,toKeep[i].Nama_Siswa,toKeep[i].JK,toKeep[i].Kelas_Asal,toKeep[i].Nilai]); }
    for (var i = 0; i < data.records.length; i++) { var r = data.records[i]; sh.appendRow([uuid_(), data.kelasId, r.Kelompok, r.No || '', r.Siswa_ID || '', r.NIS || '', r.Nama_Siswa || '', r.JK || '', r.Kelas_Asal || '', r.Nilai || '']); }
    SpreadsheetApp.flush();
    addLog_(s.userId, 'SAVE_KELOMPOK', 'Simpan kelompok ' + data.kelasId);
    return { ok: true, msg: 'Kelompok belajar berhasil disimpan' };
  } catch (e) { return { ok: false, msg: e.message }; }
}

function apiGetLCKH(t, fl) {
  var s = getSession_(t); if (!s) return { ok: false, msg: 'Unauthorized' };
  try {
    var l = getAllRows_('LCKH');
    if (fl) { if (fl.bulan) l = l.filter(function(r){ var d=String(r.Tanggal); return d.indexOf('/'+fl.bulan+'/')>=0||d.indexOf('-'+fl.bulan+'-')>=0 }); if (fl.tahun) l = l.filter(function(r){ var d=String(r.Tanggal); return d.indexOf('/'+fl.tahun)>=0||d.indexOf(fl.tahun)>=0 }) }
    l.sort(function(a,b){ var da=a.Tanggal||'',db=b.Tanggal||''; if(da<db) return -1; if(da>db) return 1; return 0 });
    return { ok: true, data: l };
  } catch (e) { return { ok: false, msg: e.message }; }
}

function parseDateAny_(s){if(!s)return null;var d=null;if(s.indexOf('-')>=0){var p=s.split('-');if(p.length===3)d=new Date(p[0]+'-'+p[1]+'-'+p[2])}else if(s.indexOf('/')>=0){var p=s.split('/');if(p.length===3)d=new Date(p[2]+'-'+p[1]+'-'+p[0])}if(d&&!isNaN(d.getTime()))return d;return null}
function apiGenerateLCKH(t, bulan, tahun) {
  var s = getSession_(t); if (!s) return { ok: false, msg: 'Unauthorized' };
  try {
    var jurnal = getAllRows_('Jurnal_Mengajar'), kl = getAllRows_('Data_Kelas');
    var kelasMap = {}; for (var i = 0; i < kl.length; i++) kelasMap[kl[i].ID] = kl[i].Nama_Kelas;
    var groups = {};
    for (var i = 0; i < jurnal.length; i++) {
      var j = jurnal[i], d = parseDateAny_(String(j.Tanggal||'')); if (!d) continue;
      var b = String(d.getMonth()+1).padStart(2,'0'), th = d.getFullYear();
      if (bulan && String(b) !== String(bulan).padStart(2,'0')) continue;
      if (tahun && String(th) !== String(tahun)) continue;
      var kn = kelasMap[j.Kelas_ID] || j.Nama_Kelas || '-';
      var tglKey = Utilities.formatDate(d, 'Asia/Jakarta', 'yyyy-MM-dd');
      var mapel = j.Mata_Pelajaran||'-';
      var key = tglKey+'|'+mapel;
      if (!groups[key]) groups[key] = { tgl: tglKey, mapel: mapel, kelas: [], jids: [] };
      if (groups[key].kelas.indexOf(kn)<0) groups[key].kelas.push(kn);
      if (j.ID && groups[key].jids.indexOf(j.ID)<0) groups[key].jids.push(j.ID);
    }
    var records = []; var keys = Object.keys(groups).sort();
    for (var i = 0; i < keys.length; i++) {
      var g = groups[keys[i]];
      records.push({
        No: i+1, Kegiatan: 'Mengajar',
        Pekerjaan: 'KBM di kelas '+g.kelas.join(', ')+' mapel '+g.mapel,
        Tanggal: g.tgl,
        Jurnal_ID: g.jids.join('|')
      });
    }
    return { ok: true, data: records };
  } catch (e) { return { ok: false, msg: e.message }; }
}

function apiGenerateLKB(t, bulan, tahun) {
  var s = getSession_(t); if (!s) return { ok: false, msg: 'Unauthorized' };
  try {
    var lckh = getAllRows_('LCKH');
    var grouped = {};
    for (var i = 0; i < lckh.length; i++) {
      var r = lckh[i], tgl = String(r.Tanggal||'');
      var d = parseDateAny_(tgl); if (!d) continue;
      var b = String(d.getMonth()+1).padStart(2,'0'), th = d.getFullYear();
      if (bulan && String(b) !== String(bulan).padStart(2,'0')) continue;
      if (tahun && String(th) !== String(tahun)) continue;
      var key = (r.Kegiatan||'')+'|'+(r.Pekerjaan||'');
      if (!grouped[key]) grouped[key] = { Kegiatan: r.Kegiatan||'', Pekerjaan: r.Pekerjaan||'', count: 0 };
      grouped[key].count++;
    }
    var records = []; var keys = Object.keys(grouped);
    for (var i = 0; i < keys.length; i++) {
      var g = grouped[keys[i]];
      records.push({
        No: i+1, Uraian_Tugas: g.Kegiatan+' '+g.Pekerjaan,
        Vol: g.count, Bukti_Dokumen: 'Dokumentasi kegiatan dan presensi',
        Bulan: bulan||'', Tahun: tahun||''
      });
    }
    return { ok: true, data: records };
  } catch (e) { return { ok: false, msg: e.message }; }
}

// === SERVE HTML & API ===
function doGet(e) {
  var action = e && e.parameter && e.parameter.action;
  if (action) {
    var token = e.parameter.token || '';
    var payload = {};
    try { payload = JSON.parse(e.parameter.payload || '{}'); } catch (x) {}
    var result = apiRouter_(action, token, payload);
    return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON);
  }
  var html = HtmlService.createHtmlOutputFromFile('Index');
  html.setTitle('Teacher Dashboard')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL)
    .addMetaTag('viewport', 'width=device-width, initial-scale=1');
  return html;
}

function doPost(e) {
  try {
    var body = JSON.parse(e.postData.contents || '{}');
    var action = body.action || '';
    var token = body.token || '';
    var payload = body.data || {};
    var result = apiRouter_(action, token, payload);
    return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON);
  } catch (x) {
    return ContentService.createTextOutput(JSON.stringify({ ok: false, msg: x.message })).setMimeType(ContentService.MimeType.JSON);
  }
}
